cmake_minimum_required(VERSION 3.20)
# This file also supports a standalone subproject build.
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  project(cuda_kernel_optimization LANGUAGES CXX)
  include(CTest)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  option(PARALLEL_ENABLE_CUDA "Build CUDA kernels when a CUDA compiler is available" ON)
  option(PARALLEL_ENABLE_GPU_TESTS "Register correctness tests requiring an NVIDIA GPU" OFF)
endif()
add_library(cpu_reference src/reference/reference.cpp)
target_compile_features(cpu_reference PUBLIC cxx_std_17)
set_target_properties(cpu_reference PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(cpu_reference PUBLIC include)
add_executable(reference_test tests/reference_test.cpp)
target_link_libraries(reference_test PRIVATE cpu_reference)
add_executable(tile_layout_test tests/tile_layout_test.cpp)
target_link_libraries(tile_layout_test PRIVATE cpu_reference)
if(BUILD_TESTING)
  add_test(NAME reference COMMAND reference_test)
  set_tests_properties(reference PROPERTIES LABELS "cpu;cuda-host")
  add_test(NAME cuda_tile_layout COMMAND tile_layout_test)
  set_tests_properties(cuda_tile_layout PROPERTIES LABELS "cpu;cuda-host")
  if(Python3_Interpreter_FOUND)
    add_test(NAME cuda_analysis COMMAND ${Python3_EXECUTABLE} -m unittest discover
      -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_*.py")
    set_tests_properties(cuda_analysis PROPERTIES LABELS "cpu;cuda-host;analysis" ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  endif()
endif()
if(PARALLEL_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
endif()
if(PARALLEL_ENABLE_CUDA AND CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_library(kernels src/kernels/baseline.cu src/kernels/tiled.cu
    src/kernels/coalesced_tiled.cu src/kernels/register_tiled.cu)
  set_target_properties(kernels PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_include_directories(kernels PUBLIC include)
  target_compile_options(kernels PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
  add_executable(kernel_benchmark src/main.cu)
  set_target_properties(kernel_benchmark PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(kernel_benchmark PRIVATE kernels cpu_reference)
  add_executable(correctness_test tests/correctness_test.cu)
  set_target_properties(correctness_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(correctness_test PRIVATE kernels cpu_reference)
  if(BUILD_TESTING AND PARALLEL_ENABLE_GPU_TESTS)
    add_test(NAME gpu_correctness COMMAND correctness_test)
    set_tests_properties(gpu_correctness PROPERTIES LABELS "gpu;cuda")
  else()
    message(STATUS "GPU runtime tests are disabled; enable PARALLEL_ENABLE_GPU_TESTS on an NVIDIA host")
  endif()
else()
  message(STATUS "CUDA unavailable or disabled: host reference and analysis remain available; GPU targets are not built")
endif()
