cmake_minimum_required(VERSION 3.20)
project(parallel_computing_lab LANGUAGES CXX)
include(CTest)
option(PARALLEL_ENABLE_THREADS "Build the C++ thread synchronization experiment" ON)
option(PARALLEL_ENABLE_OPENMP "Build the OpenMP scheduling experiment when supported" ON)
option(PARALLEL_BUILD_CUDA_SUBPROJECT "Build the CUDA subproject, including its host-only reference" ON)
option(PARALLEL_ENABLE_CUDA "Build CUDA kernels when a CUDA compiler is available" ON)
option(PARALLEL_ENABLE_GPU_TESTS "Register correctness tests requiring an NVIDIA GPU" OFF)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(PARALLEL_ENABLE_THREADS)
  find_package(Threads REQUIRED)
  add_executable(pi_mc cpp-multithreading/src/main.cpp)
  target_compile_features(pi_mc PRIVATE cxx_std_11)
  set_target_properties(pi_mc PROPERTIES CXX_EXTENSIONS OFF)
  target_link_libraries(pi_mc PRIVATE Threads::Threads)
endif()
if(PARALLEL_ENABLE_OPENMP)
  find_package(OpenMP COMPONENTS CXX QUIET)
  if(OpenMP_CXX_FOUND)
    add_executable(scheduling openmp-scheduling/src/main.cpp)
    target_compile_features(scheduling PRIVATE cxx_std_11)
    set_target_properties(scheduling PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(scheduling PRIVATE OpenMP::OpenMP_CXX)
  else()
    message(STATUS "OpenMP not found: scheduling target and tests are unavailable")
  endif()
endif()
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME benchmark_tools COMMAND ${Python3_EXECUTABLE} -m unittest discover
    -s ${CMAKE_CURRENT_SOURCE_DIR}/benchmarks/tests -p "test_*.py")
  set_tests_properties(benchmark_tools PROPERTIES LABELS "cpu;benchmarks" ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  if(TARGET pi_mc)
    add_test(NAME monte_carlo COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/cpp-multithreading/tests/test_cli.py $<TARGET_FILE:pi_mc>)
    set_tests_properties(monte_carlo PROPERTIES LABELS "cpu;threads" ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  endif()
  if(TARGET scheduling)
    add_test(NAME scheduling COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/openmp-scheduling/tests/test_cli.py $<TARGET_FILE:scheduling>)
    set_tests_properties(scheduling PROPERTIES LABELS "cpu;openmp" ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
  endif()
endif()
if(PARALLEL_BUILD_CUDA_SUBPROJECT)
  add_subdirectory(cuda-kernel-optimization)
endif()
if(BUILD_TESTING AND NOT Python3_Interpreter_FOUND)
  message(STATUS "Python 3 not found: CLI and analysis tests are unavailable")
endif()
